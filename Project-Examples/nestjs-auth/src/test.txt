# 1. Register
curl -X POST http://localhost:3000/api/auth/register \
  -H "Content-Type: application/json" \
  -d '{
    "email": "test@example.com",
    "password": "password123"
  }'

# 2. Login (accessToken sẽ được set vào cookie)
curl -X POST http://localhost:3000/api/auth/login \
  -H "Content-Type: application/json" \
  -c cookies.txt \
  -d '{
    "email": "admin@example.com",
    "password": "password123"
  }'

# Response sẽ có dạng:
# {
#   "message": "Login successful",
#   "refreshToken": "eyJhbGciOiJIUzI1NiIsInR5cCI6IkpXVCJ9...",
#   "user": {
#     "id": "uuid",
#     "email": "admin@example.com"
#   }
# }

# 3. Access protected route (cookie sẽ tự động gửi)
curl -X GET http://localhost:3000/api/users/profile \
  -b cookies.txt

# 4. Refresh token
curl -X POST http://localhost:3000/api/auth/refresh \
  -H "Content-Type: application/json" \
  -c cookies.txt \
  -d '{
    "refreshToken": "YOUR_REFRESH_TOKEN_HERE"
  }'

# 5. Test RBAC - Admin only route
curl -X GET http://localhost:3000/api/users \
  -b cookies.txt

# 6. Test permission-based route
curl -X POST http://localhost:3000/api/posts \
  -H "Content-Type: application/json" \
  -b cookies.txt \
  -d '{
    "title": "Test Post",
    "content": "This is a test post"
  }'

# 7. Logout
curl -X POST http://localhost:3000/api/auth/logout \
  -b cookies.txt

# 8. Revoke all tokens (khi phát hiện token bị đánh cắp)
curl -X POST http://localhost:3000/api/auth/revoke-all \
  -b cookies.txt
